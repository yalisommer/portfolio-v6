    ```ts
    function applyMouseRepulsion(
      objects: BouncingMesh[],
      mouseScreenX: number,
      mouseScreenY: number,
      camera: THREE.PerspectiveCamera,
      bounds: Bounds,
    ) {
      // Convert screen coords to Three.js world coords
      const worldX = (mouseScreenX / window.innerWidth  - 0.5) * bounds.w
      const worldY = camera.position.y + (0.5 - mouseScreenY / window.innerHeight) * bounds.h

      for (const obj of objects) {
        const dx = obj.mesh.position.x - worldX
        const dy = obj.mesh.position.y - worldY
        const dist = Math.sqrt(dx * dx + dy * dy)

        if (dist < REPULSION_RADIUS && dist > 0.001) {
          const strength = (1 - dist / REPULSION_RADIUS) * REPULSION_STRENGTH
          const nx = dx / dist
          const ny = dy / dist
          obj.velocity.x += nx * strength
          obj.velocity.y += ny * strength

          // Clamp to repulsion max
          const speed = Math.sqrt(obj.velocity.x * obj.velocity.x + obj.velocity.y * obj.velocity.y)
          if (speed > VELOCITY_MAX_REPULSION) {
            const scale = VELOCITY_MAX_REPULSION / speed
            obj.velocity.x *= scale
            obj.velocity.y *= scale
          }
        }
      }
    }
    ```

    **Step 4 — Call `applyMouseRepulsion` inside the `animate` function**, immediately before `updatePhysics`:
