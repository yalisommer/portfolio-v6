```typescript
import { useState } from 'react'
import AquariumLanding from './components/AquariumLanding'
import Nav from './components/Nav'
import Section from './components/Section'
import MeshBackground from './components/MeshBackground'
import AboutSection from './components/sections/AboutSection'
import ExperienceSection from './components/sections/ExperienceSection'
import EducationSection from './components/sections/EducationSection'
import SkillsSection from './components/sections/SkillsSection'
import ProjectsSection from './components/sections/ProjectsSection'
import ResearchSection from './components/sections/ResearchSection'
import ContactSection from './components/sections/ContactSection'

const SECTIONS: { id: string; Content: () => JSX.Element }[] = [
  { id: 'about', Content: AboutSection },
  { id: 'experience', Content: ExperienceSection },
  { id: 'education', Content: EducationSection },
  { id: 'skills', Content: SkillsSection },
  { id: 'projects', Content: ProjectsSection },
  { id: 'research', Content: ResearchSection },
  { id: 'contact', Content: ContactSection },
]

export default function App() {
  const [heroVisible, setHeroVisible] = useState(true)

  return (
    <>
      <Nav heroVisible={heroVisible} />
      <AquariumLanding onHeroVisibility={setHeroVisible} heroVisible={heroVisible} />
      <MeshBackground active={!heroVisible} />
      <div style={{ position: 'relative', zIndex: 10, paddingTop: '35vh' }}>
        <div style={{
          position: 'absolute',
          top: 0,
          left: 0,
          right: 0,
          height: '35vh',
          background: 'linear-gradient(to bottom, transparent 0%, #000000 100%)',
          zIndex: 2,
          pointerEvents: 'none',
        }} />
        {SECTIONS.map(({ id, Content }) => (
          <Section key={id} id={id}>
            <Content />
          </Section>
        ))}
      </div>
    </>
  )
}
```

Key points:
- **Preserve the gradient overlay div** (transparent-to-black at top of content zone) -- this is critical for the aquarium-to-content transition
- **Preserve the content zone zIndex: 10** and paddingTop: '35vh' -- established in Phase 1
- **MeshBackground placed OUTSIDE the content zone div** so it can be `position: fixed` independently. Its zIndex: 5 puts it above the aquarium (0) but below the content zone (10).
- **Remove the `DS` import** from App.tsx since it's no longer used there (section components import it themselves). BUT double-check: if the gradient div or any remaining inline style uses DS.*, keep the import. In the current code, the gradient uses hardcoded `#000000` and `transparent`, NOT DS.*, so the import can be removed.
- **Remove the `SECTION_IDS` const** -- replaced by the SECTIONS array.
- The SECTIONS array typing: use `() => JSX.Element` for the Content field, or use `React.ComponentType` if preferred. Either works.
